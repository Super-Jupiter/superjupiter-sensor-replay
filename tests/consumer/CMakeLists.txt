cmake_minimum_required(VERSION 3.16)
project(SuperJupiterSensorReplayConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_SENSOR_REPLAY_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_SENSOR_REPLAY_SOURCE_DIR}" sensor-replay)
  if(TARGET superjupiter-sensor-replay-tests OR TARGET superjupiter-sensor-replay-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterSensorReplay 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-sensor-replay-consumer main.cpp)
target_link_libraries(superjupiter-sensor-replay-consumer PRIVATE SuperJupiter::SensorReplay)
